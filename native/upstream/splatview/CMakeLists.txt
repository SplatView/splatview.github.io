cmake_minimum_required(VERSION 3.25)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "14.0" CACHE STRING "Minimum macOS version")
endif()
project(SplatViewMac VERSION 0.1.0 LANGUAGES C CXX)
if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OBJCXX_STANDARD 20)
    set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(SPLATVIEW_BUILD_MACOS "Build the native app and Quick Look extensions" ${APPLE})
option(SPLATVIEW_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(SPLATVIEW_DOMAIN_ONLY "Build only portable domain/reference renderer and contracts (no codecs or UI)" OFF)

find_package(Threads REQUIRED)
include(cmake/Concurrency.cmake)
find_package(nlohmann_json CONFIG REQUIRED)
include(CTest)
include(cmake/Domain.cmake)
if(SPLATVIEW_DOMAIN_ONLY)
    return()
endif()
if(NOT TARGET SplatView::RenderCPU)
    message(FATAL_ERROR "The full core and CLI require SPLATVIEW_BUILD_CPU_RENDERER=ON. Disable it only for a standalone shared/domain build.")
endif()
include(cmake/VerifySpz.cmake)
splatview_verify_spz("${CMAKE_CURRENT_SOURCE_DIR}")
find_package(ZLIB REQUIRED)
find_package(zstd CONFIG REQUIRED)
find_package(libdeflate CONFIG QUIET)
if(NOT TARGET libdeflate::libdeflate_static)
    find_path(LIBDEFLATE_INCLUDE_DIR libdeflate.h REQUIRED)
    find_library(LIBDEFLATE_LIBRARY NAMES deflate deflatestatic REQUIRED)
    add_library(libdeflate::libdeflate_static UNKNOWN IMPORTED)
    set_target_properties(libdeflate::libdeflate_static PROPERTIES IMPORTED_LOCATION "${LIBDEFLATE_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${LIBDEFLATE_INCLUDE_DIR}")
endif()
find_package(WebP CONFIG QUIET)
if(NOT TARGET WebP::webp)
    find_path(WEBP_INCLUDE_DIR webp/decode.h REQUIRED)
    find_library(WEBP_LIBRARY NAMES webp libwebp REQUIRED)
    add_library(WebP::webp UNKNOWN IMPORTED)
    set_target_properties(WebP::webp PROPERTIES IMPORTED_LOCATION "${WEBP_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${WEBP_INCLUDE_DIR}")
endif()
set(SPZ_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(SPZ_BUILD_PYTHON_BINDINGS OFF CACHE BOOL "" FORCE)
set(SPZ_BUILD_EXTENSIONS ON CACHE BOOL "" FORCE)
add_subdirectory(third_party/spz EXCLUDE_FROM_ALL)

file(READ "${CMAKE_CURRENT_SOURCE_DIR}/src/core/gsmod_presets.json" GSMOD_PRESET_JSON)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/core/gsmod_presets.json)
configure_file(src/core/gsmod_presets.hpp.in generated/gsmod_presets.hpp @ONLY)
add_library(splatview_core STATIC
    src/core/model.cpp src/core/extra_formats.cpp src/core/ply_preview.cpp
    src/core/ply_export.cpp src/core/processing.cpp
    src/core/scene_exports.cpp src/core/extra_exports.cpp src/core/sog_export.cpp
    src/platform/file_io.cpp src/platform/atomic_file.cpp src/platform/image_io.cpp
    src/platform/file_preferences.cpp)
target_sources(splatview_core PRIVATE src/core/document.cpp src/core/background_jobs.cpp src/core/preview_coordinator.cpp)
target_sources(splatview_core PRIVATE src/platform/source_snapshot.cpp src/platform/preference_job.cpp)
target_sources(splatview_core PRIVATE src/core/language.cpp)
target_sources(splatview_core PRIVATE src/platform/attached_view.cpp)
target_sources(splatview_core PRIVATE src/platform/preference_lock.cpp src/core/tools_preferences.cpp)
if(APPLE)
    target_sources(splatview_core PRIVATE macos/Shared/image_io_macos.mm)
    set_source_files_properties(macos/Shared/image_io_macos.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(splatview_core PUBLIC "-framework CoreFoundation" "-framework Foundation" "-framework ImageIO" "-framework CoreGraphics")
endif()
target_include_directories(splatview_core PUBLIC src/core src/platform src/render "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(splatview_core SYSTEM PUBLIC third_party/DirectXMath/Inc)
target_include_directories(splatview_core SYSTEM PRIVATE third_party/stb third_party/picosha2)
target_compile_definitions(splatview_core PUBLIC _XM_NO_XMVECTOR_OVERLOADS_ $<$<CXX_COMPILER_ID:MSVC>:_ENABLE_EXTENDED_ALIGNED_STORAGE>)
target_link_libraries(splatview_core PUBLIC SplatView::RenderCPU Threads::Threads spz::spz ZLIB::ZLIB libdeflate::libdeflate_static WebP::webp nlohmann_json::nlohmann_json)
if(MSVC)
    target_compile_options(splatview_core PRIVATE /utf-8 /W4 /permissive- /EHsc /fp:precise)
    target_compile_definitions(splatview_core PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
else()
    target_compile_options(splatview_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    if(SPLATVIEW_ENABLE_SANITIZERS)
        target_compile_options(splatview_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(splatview_core PUBLIC -fsanitize=address,undefined)
    endif()
endif()

add_executable(splatview-cli src/cli/main.cpp)
target_link_libraries(splatview-cli PRIVATE splatview_core)
if(MSVC)
    target_compile_options(splatview-cli PRIVATE /utf-8)
endif()
install(TARGETS splatview-cli RUNTIME DESTINATION bin)
install(FILES LICENSE THIRD-PARTY-NOTICES.txt THIRD_PARTY_PORT_NOTICES.txt DESTINATION share/splatview/licenses)
install(FILES third_party/DirectXMath/LICENSE DESTINATION share/splatview/licenses RENAME DirectXMath.txt)
install(FILES third_party/DirectXMath/SAL-LICENSE.txt DESTINATION share/splatview/licenses)
install(FILES third_party/stb/LICENSE DESTINATION share/splatview/licenses RENAME stb.txt)
install(FILES third_party/picosha2/LICENSE DESTINATION share/splatview/licenses RENAME PicoSHA2.txt)
install(FILES third_party/imgui/LICENSE.txt DESTINATION share/splatview/licenses RENAME imgui.txt)
install(FILES third_party/ImGuizmo/LICENSE DESTINATION share/splatview/licenses RENAME ImGuizmo.txt)
install(FILES third_party/spz/LICENSE DESTINATION share/splatview/licenses RENAME spz.txt)
install(FILES third_party/spz/LOCAL_CHANGES.md third_party/spz/LOCAL_PATCHES.json
    DESTINATION share/splatview/licenses/spz-patch)
install(DIRECTORY third_party/format-reference-licenses/ DESTINATION share/splatview/licenses/format-references)

add_library(splatview_imgui STATIC third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
    third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp third_party/ImGuizmo/ImGuizmo.cpp)
target_include_directories(splatview_imgui SYSTEM PUBLIC third_party/imgui third_party/ImGuizmo)
add_library(splatview_ui STATIC src/ui/editor_ui.cpp)
target_include_directories(splatview_ui PUBLIC src/ui)
target_link_libraries(splatview_ui PUBLIC splatview_core splatview_imgui)
if(MSVC)
    target_compile_options(splatview_ui PRIVATE /utf-8)
endif()

    add_executable(SplatViewCoreTests tests/core_contracts.cpp tests/format_fixtures.cpp tests/io_contracts.cpp
        tests/export_contracts.cpp tests/processing_contracts.cpp tests/geometry_contracts.cpp tests/ply_preview_contracts.cpp tests/portable_contracts.cpp tests/format_decode_contracts.cpp tests/surface_contracts.cpp tests/camera_contracts.cpp tests/document_job_contracts.cpp tests/preview_kernel_contracts.cpp
        tests/document_draft_contracts.cpp tests/background_job_contracts.cpp tests/frame_capture_contracts.cpp
        tests/document_loader_contracts.cpp tests/image_output_contracts.cpp
        tests/resource_budget_contracts.cpp tests/preference_job_contracts.cpp
        tests/source_admission_contracts.cpp tests/scene_storage_contracts.cpp
        tests/decode_admission_contracts.cpp tests/decode_work_contracts.cpp tests/side_input_contracts.cpp
        tests/image_decode_contracts.cpp tests/document_capture_contracts.cpp
        tests/spz_admission_contracts.cpp tests/admitted_storage_contracts.cpp
        tests/history_admission_contracts.cpp tests/processing_scratch_contracts.cpp
        tests/selection_work_contracts.cpp tests/bounds_work_contracts.cpp
        tests/selection_map_work_contracts.cpp tests/preview_coordinator_contracts.cpp
        tests/render_storage_contracts.cpp tests/renderer_cache_contracts.cpp)
    target_compile_definitions(SplatViewCoreTests PRIVATE FORMAT_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
    target_link_libraries(SplatViewCoreTests PRIVATE splatview_core)
    if(MSVC)
        target_compile_options(SplatViewCoreTests PRIVATE /utf-8 /fp:precise)
    endif()
    add_test(NAME CoreContracts COMMAND SplatViewCoreTests "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
    set_tests_properties(CoreContracts PROPERTIES TIMEOUT 180)
    add_executable(SplatViewUiTests tests/ui_contracts.cpp)
    target_link_libraries(SplatViewUiTests PRIVATE splatview_ui)
    if(MSVC)
        target_compile_options(SplatViewUiTests PRIVATE /utf-8)
    endif()
    add_test(NAME UiContracts COMMAND SplatViewUiTests "${CMAKE_CURRENT_BINARY_DIR}/ui-fixtures")
    set_tests_properties(UiContracts PROPERTIES TIMEOUT 60)
    # The full suite includes CLI, bundle and provenance contracts. A release
    # lane must not silently omit them when Python is absent.
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    if(Python3_Interpreter_FOUND)
        add_test(NAME CliContracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_contracts.py"
            "$<TARGET_FILE:splatview-cli>" "${CMAKE_CURRENT_BINARY_DIR}/cli-fixtures")
        add_test(NAME BundleConfiguration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/check_bundle_config.py" --cmake "${CMAKE_COMMAND}")
        add_test(NAME VendorIntegrityContracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/vendor_integrity_contracts.py"
            "${CMAKE_COMMAND}" "${CMAKE_GENERATOR}" "${CMAKE_MAKE_PROGRAM}")
        set_tests_properties(VendorIntegrityContracts PROPERTIES TIMEOUT 60 LABELS "portable;provenance")
        add_test(NAME SourceCaptureContracts COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/source_capture_contracts.py")
        set_tests_properties(SourceCaptureContracts PROPERTIES TIMEOUT 60 LABELS "portable;provenance")
        set_tests_properties(CliContracts PROPERTIES TIMEOUT 180)
    endif()
endif()
if(SPLATVIEW_BUILD_MACOS)
    if(NOT APPLE)
        message(FATAL_ERROR "The native app needs the macOS SDK. Build the core and CLI on this platform instead.")
    endif()
    add_subdirectory(macos)
endif()
